#### OSKAR ####
wget https://github.com/OxfordSKA/OSKAR/archive/refs/tags/2.8.3.tar.gz
tar vxfz 2.8.3.tar.gz
cd OSKAR-2.8.3
mkdir build && cd build
mkdir ~/oskar-bin 2> /dev/null
cmake -DCMAKE_INSTALL_PREFIX=~/oskar-bin -DFIND_CUDA=OFF ../
make -j8
make install
export OSKAR_INC_DIR=~/oskar-bin/include
export OSKAR_LIB_DIR=~/oskar-bin/lib
pip install --use-pep517 'git+https://github.com/OxfordSKA/OSKAR.git@master#egg=oskarpy&subdirectory=python
pip install ska-sdp-datamodels --extra-index-url https://artefact.skao.int/repository/pypi-internal/simple
